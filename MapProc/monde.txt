# Monde entier tiré au hasard (plugin MapProceduralGenerator) : villes reliées
# par des routes, chaque ville un peu plus loin du départ, donc les arènes dans
# cet ordre. Chaque ville vient de ville.txt, chaque route de route.txt.
# Lancer : menu debug > Autres éditeurs > Générer un lieu (MapProc) > [Monde] monde.txt.
# MapGen écrit ensuite les raccords, les métadonnées des cartes et la Carte de
# la région (PBS/map_connections.txt, map_metadata.txt, town_map.txt).
Monde    = TypeLocked
Region   = 0
Image    = typelocked_proc.png
Origin   = 1,1
Taille   = 28x18
Cell     = 20
Ville    = ville.txt
Route    = route.txt
Villes   = 9
Arenes   = 8
# La première ville n'a jamais d'arène. Les autres villes sans arène sont tirées
# au hasard dans la chaîne, ou imposées par leur numéro, ex. « SansArene = 5 ».
Longueur = 2..3
# L'eau qui barre une route (chenal tout droit ou lac d'un bord à l'autre) ne se
# traverse qu'avec Surf : elle n'apparaît qu'après la ville de la 2e arène.
# Les lacs posés dans les villes et les routes ne bloquent jamais le passage.
Surf     = 2
Chenaux  = 1..2
Lacs     = 0..1
# Routes barrées par un mur de rochers, que l'on passe par une grotte.
Grottes  = 1..2
# Routes de montagne : on y monte et descend de niveau en niveau par des
# escaliers. Une route de montagne peut aussi être coupée par un chenal ou un
# lac. Certaines ont une grotte, un cul-de-sac à explorer (en %).
Montagnes = 1..2
GrotteMontagne = 50
# Plongée s'obtient dans une ville à arène, la 3e ou une suivante, et toujours
# après la ville de Surf (il faut Surf pour plonger). Les zones sous-marines ne
# sont posées que sur des routes d'eau situées après la ville de Plongée : on y
# plonge depuis l'eau profonde.
Plongee  = 3
Fonds    = 1
# Démesure : chance (en %) qu'un lieu soit hors norme : forêt profonde, mer
# d'herbes, chaos rocheux, lac géant, montagne colossale, route interminable,
# ville-jardin, cité lacustre, grande foire.
Demesure = 25
