# Monde entier tiré au hasard (plugin MapProceduralGenerator) : villes reliées
# par des routes, chaque ville un peu plus loin du départ, donc les arènes dans
# cet ordre. Chaque ville vient de ville.txt, chaque route de route.txt.
# Lancer : menu debug > Autres éditeurs > Générer un lieu (MapProc) > [Monde] monde.txt.
# MapGen écrit ensuite les raccords, les métadonnées des cartes et la Carte de
# la région (PBS/map_connections.txt, map_metadata.txt, town_map.txt).
Monde    = TypeLocked
Region   = 0
Image    = typelocked_proc.png
# 32x24 cases : tout l'écran de la carte de la région (512x384, cases de 16).
Origin   = 0,0
Taille   = 32x24
Cell     = 20
# Marge : cases laissées libres au bord de la grille, pour que la côte de la
# carte de région ne soit pas coupée par le bord de l'écran (0 par défaut).
Marge    = 1
Ville    = ville.txt
Route    = route.txt
# Noms affichés en jeu (villes, grottes, région), tirés avec la graine. Les
# cartes gardent leur nom généré dans l'arbre des cartes.
Toponymie = Lore/noms.txt
Villes   = 11
Arenes   = 8
# La première ville n'a jamais d'arène. Les autres villes sans arène sont tirées
# au hasard dans la chaîne, ou imposées par leur numéro, ex. « SansArene = 5 ».
# 11 villes pour 8 arènes : deux villes sans arène en plus de la première,
# qui peuvent être des villages (1x1, voir TaillesVilles).
# Routes de 1 à 4 cases de long, de 1 ou 2 cases de large (herbes, eau et
# grottes s'étalent sur toute la largeur).
Longueur = 1..4
Largeur  = 1..2
# Tailles des villes, en cases : 1x1 (village, jamais d'arène), 1x2 (ou 2x1),
# 2x2 et L (3 cases d'un carré 2x2, les rues passent par le coude). La première
# ville fait au moins 1x2 (le labo du Professeur).
TaillesVilles = 1x1, 1x2, 2x2, L
# Formes des routes : droites, en L (un virage ; au moins la largeur + 2 cases
# de long) ou en U (elle revient vers une ville posée à côté de celle qu'elle
# quitte, du même côté ; des branches de sa longueur).
# Une route en L garde tout son rectangle sur la grille du monde ; les cases
# hors de son tracé sont remplies d'arbres.
FormesRoutes = droite, L, U
# Boucles : routes de plus entre deux zones de la chaîne sans ville à arène
# entre elles, ouvertes dès le début (0 = aucune). Une route ajoutée fait
# jusqu'à 6 cases de long pour contourner la chaîne.
Boucles  = 1..3
# Raccourcis : routes entre deux zones plus éloignées, fermées du côté du départ
# par un obstacle (arbre à couper, rocher fissuré, rocher à pousser, eau à
# surfer). L'objet qui l'ouvre s'obtient au plus tôt dans la dernière ville à
# arène avant l'autre bout : un raccourci ne fait jamais sauter d'arène.
# Ce qui ne trouve pas de place est écrit dans la console (0 = aucun).
Raccourcis = 1..3
# L'eau qui barre une route (chenal tout droit ou lac d'un bord à l'autre) ne se
# traverse qu'avec Surf : elle n'apparaît qu'après la ville de la 2e arène.
# Les lacs posés dans les villes et les routes ne bloquent jamais le passage.
# Mers : routes presque entièrement sur la mer (plages aux deux bouts, crêtes et
# massifs de roche, bancs de sable), comme les routes marines de Hoenn.
Surf     = 2
Chenaux  = 1..2
Lacs     = 0..1
Mers     = 1..2
# Routes barrées par un mur de rochers, que l'on passe par une grotte.
Grottes  = 1..2
# Routes de montagne : on y monte et descend de niveau en niveau par des
# escaliers. Une route de montagne peut aussi être coupée par un chenal ou un
# lac. Certaines ont une grotte, un cul-de-sac à explorer (en %).
Montagnes = 1..2
GrotteMontagne = 50
# Une route de montagne nord-sud peut avoir une rivière qui longe la montée,
# au pied des falaises, en parallèle du chemin (en %).
RiviereMontagne = 50
# Villes portuaires : la mer sur un côté sans sortie, un quai, des pontons et
# une gare maritime sur le quai. Une ville nommée Port, Dock ou Harbour en est
# toujours une, en plus de celles-ci.
Ports    = 1..2
# Un port avec une gare maritime (mer au sud, à l'est ou à l'ouest) a, selon
# ce pourcentage, un bateau vers une île : une carte à part, village ou île
# sauvage, nommée « Île … » ou comme une ville.
Iles     = 50
# Une île peut avoir sa propre arène (en %), le port gardant la sienne : l'île
# est alors un village. Le nombre d'arènes ne change pas : une ville plus loin
# dans la chaîne (sinon le port) cède la sienne, jamais celle de Surf ou de
# Plongée.
IlesArene = 50
# Plongée s'obtient dans une ville à arène, la 3e ou une suivante, et toujours
# après la ville de Surf (il faut Surf pour plonger). Les zones sous-marines ne
# sont posées que sur des routes d'eau situées après la ville de Plongée : on y
# plonge depuis l'eau profonde.
Plongee  = 3
Fonds    = 1
# Objets de terrain : chacun s'obtient dans une ville à arène, celle de la
# n-ième arène ou la suivante, dans l'ordre de ces numéros (Surf et Plongée
# ont leurs clés plus haut).
Machette = 1
Lanterne = 2
Pioche   = 3
Harnais  = 4
Grappin  = 5
Vol      = 6
# Recoins : chance (en %) qu'un lieu ait un recoin facultatif, fermé par un
# obstacle qu'ouvre un objet obtenu plus tôt : arbre à couper, rocher fissuré,
# rocher à pousser, îlot dans un étang (Surf), grotte sombre (Lanterne).
Recoins  = 30
# Démesure : chance (en %) qu'un lieu soit hors norme : forêt profonde, mer
# d'herbes, chaos rocheux, lac géant, montagne colossale, route interminable,
# ville-jardin, cité lacustre, grande foire, métropole ; et hors de la chaîne, une île
# (comme une ville ou une route), une grotte (grotte immense, dédale) ou des
# fonds (abysse, forêt d'algues).
Demesure = 75
