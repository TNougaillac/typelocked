# Route assemblée à partir des briques de MapProc/Briques/ (plugin MapProceduralGenerator).
# Dans un monde (monde.txt), la taille et les sorties sont données par le monde.
# Les quantités « par case » sont comptées pour 20x20 tuiles : une route plus
# longue a plus d'herbes et d'arbres.
Lieu    = route
Nom     = Route générée
Tileset = 1
Taille  = 20x60
Sol     = .
Rue     = .
Bordure = t
Sorties = nord sud
Rues    = 0..1
Eau     = ~
Profond = %
Roche   = R
Caillou = o
# Routes de montagne (« Montagnes » dans monde.txt) : niveaux 1 et 2, puis les
# escaliers (2 de large) : face sud et rebord nord du niveau 1, puis du niveau 2.
Relief   = ^ &
Escalier = ef gh EF GH

[Briques]
prairie = 1
herbes = 0..2 par case
arbre  = 1..3 par case
parc   = 0..1 par case

corniche = 0..1 par case
# Rochers sur les niveaux d'une route de montagne (seulement là).
rocaille = 2..5 par case

# Entrées de grotte : posées seulement sur les routes que le monde fait passer
# par une grotte (« Grottes » dans monde.txt).
grotte = 0

[Ambiance prairie]
