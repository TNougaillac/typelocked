# Route assemblée à partir des briques de MapProc/Briques/ (plugin MapProceduralGenerator).
# Dans un monde (monde.txt), la taille et les sorties sont données par le monde.
# Les quantités « par case » sont comptées pour 20x20 tuiles : une route plus
# longue a plus d'herbes et d'arbres.
Lieu    = route
Nom     = Route générée
Tileset = 1
Taille  = 20x60
Sol     = .
Rue     = .
Bordure = t
Sorties = nord sud
Rues    = 0..1
Eau     = ~
Profond = %
Roche   = R
Caillou = o

[Briques]
prairie = 1
herbes = 0..2 par case
arbre  = 1..3 par case
parc   = 0..1 par case
lac    = 0..1
# Entrées de grotte : posées seulement sur les routes que le monde fait passer
# par une grotte (« Grottes » dans monde.txt).
grotte = 0

# Ambiances : chaque route en tire une selon son poids (l'herbe le plus souvent).
# « {sol} » dans la palette = la première tuile du sol tiré.
[Ambiance verdoyante]
Poids = 6
Sol   = T1|T1|T1|T2|T3

[Ambiance prairie]
Poids = 1
Sol   = T50

[Ambiance sable]
Poids = 1
Sol   = T162

[Ambiance terre battue]
Poids = 1
Sol   = T106

# Palette MapGen (tileset Outside).
[Legend]
" = T7               # hautes herbes
* = {sol} A5            # massif de fleurs
~ = {sol} A1            # eau (lacs, chenaux ; on la traverse avec Surf)
% = {sol} A1 A3         # eau profonde (on y plonge avec Plongée)
o = {sol} T815          # caillou (comble les trous du mur de rochers)
1 = T787             # rocher de l'entrée de grotte
2 = T790
3 = T789
4 = T795
5 = T798
6 = T797
7 = T803
8 = T806
9 = T805

[Objects]
t = T420 2x3         # arbre
R = T784 3x3         # rocher (mur que l'on passe par la grotte)
